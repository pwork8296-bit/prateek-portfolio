@app.websocket("/ws/ptl-rack")
async def ptl_rack_telemetry(websocket: WebSocket, manager: RackConnectionManager):
    await manager.connect(websocket)
    try:
        while True:
            event = await websocket.receive_json()
            # Broadcast LED illuminate signal to physical Modbus / MQTT controller
            await manager.broadcast_to_rack(bay_id=event["bay_id"], bin=event["bin"], action="LIGHT_ON")
    except WebSocketDisconnect:
        manager.disconnect(websocket)
